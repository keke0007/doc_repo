    你输入的用户名前缀是: $rep
    你输入的统一密码是: $pass

EOF
    read -p "你确定要创建吗? [y|n]" aredy
    case $aredy in
    y|Y|yes|YES)
    for i in $(seq $num)
    do
    username=$rep$i
    id $username &>/dev/null
    if [ $? -ne 1 ];then

    echo "$pass"|passwd --stdin $username &>/dev/null
    echo "$username Is OK!"

    done
    ;;
    n|N|No|NO)

    *)
